# vultureOS is built in "parts", each with its own toolchain:
#
#   host    tools and unit tests that run on the build machine
#   uefi    VultureBoot, the UEFI loader (PE32+)
#   target    the kernel (ELF, for the architecture being built)
#   userland  programs and libraries that run in user mode (ELF, likewise)
#
# Configuring this directory without VULTURE_PART sets up the superbuild,
# which drives every part and provides the run/test targets.

cmake_minimum_required(VERSION 3.25)

set(VULTURE_VERSION 0.1.0)

if(NOT DEFINED VULTURE_PART)
    project(vultureOS VERSION ${VULTURE_VERSION} LANGUAGES NONE)
    include(Meta/CMake/Superbuild.cmake)
    return()
endif()

project(vultureOS-${VULTURE_PART} VERSION ${VULTURE_VERSION} LANGUAGES CXX)

set(VULTURE_SOURCE_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
include(Meta/CMake/Flags.cmake)

if(VULTURE_PART STREQUAL "host")
    enable_testing()
    add_subdirectory(Libraries/LibAnimation)
    add_subdirectory(Libraries/LibGesture)
    add_subdirectory(Libraries/LibACPI)
    add_subdirectory(Libraries/LibArchive)
    add_subdirectory(Libraries/LibAudio)
    add_subdirectory(Libraries/LibBase)
    add_subdirectory(Libraries/LibCompress)
    add_subdirectory(Libraries/LibCrypto)
    add_subdirectory(Libraries/LibDeviceTree)
    add_subdirectory(Libraries/LibELF)
    add_subdirectory(Libraries/LibGfx)
    add_subdirectory(Libraries/LibHTTP)
    add_subdirectory(Libraries/LibImage)
    add_subdirectory(Libraries/LibJSON)
    add_subdirectory(Libraries/LibLocale)
    add_subdirectory(Libraries/LibPackage)
    add_subdirectory(Libraries/LibTest)
    add_subdirectory(Libraries/LibText)
    add_subdirectory(Libraries/LibTLS)
    add_subdirectory(Libraries/LibUI)
    add_subdirectory(Libraries/LibUnicode)
    add_subdirectory(Libraries/LibVT)
    add_subdirectory(Libraries/LibForeignFS)
    add_subdirectory(Libraries/LibVultureFS)
    add_subdirectory(Tests)
    add_subdirectory(Tools)
elseif(VULTURE_PART STREQUAL "uefi")
    add_subdirectory(Libraries/LibBase)
    add_subdirectory(Libraries/LibDeviceTree)
    add_subdirectory(Libraries/LibELF)
    add_subdirectory(Boot)
elseif(VULTURE_PART STREQUAL "target")
    enable_language(ASM)
    add_subdirectory(Libraries/LibACPI)
    add_subdirectory(Libraries/LibBase)
    add_subdirectory(Libraries/LibDeviceTree)
    add_subdirectory(Libraries/LibArchive)
    add_subdirectory(Libraries/LibCompress)
    add_subdirectory(Libraries/LibCrypto)
    add_subdirectory(Libraries/LibELF)
    add_subdirectory(Libraries/LibUnicode)
    add_subdirectory(Libraries/LibForeignFS)
    add_subdirectory(Libraries/LibVultureFS)
    add_subdirectory(Kernel)
    if(CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64" AND NOT VULTURE_KERNEL_SANITIZERS)
        add_subdirectory(Boot/RaspberryPi)
    endif()
elseif(VULTURE_PART STREQUAL "userland")
    enable_language(ASM)
    include(Meta/CMake/Userland.cmake)
    add_subdirectory(Libraries/LibAnimation)
    add_subdirectory(Libraries/LibGesture)
    add_subdirectory(Libraries/LibBase)
    add_subdirectory(Libraries/LibArchive)
    add_subdirectory(Libraries/LibAudio)
    add_subdirectory(Libraries/LibC)
    add_subdirectory(Libraries/LibCompress)
    add_subdirectory(Libraries/LibCore)
    add_subdirectory(Libraries/LibDesktop)
    add_subdirectory(Libraries/LibCrypto)
    add_subdirectory(Libraries/LibCxxRT)
    add_subdirectory(Libraries/LibDyld)
    add_subdirectory(Libraries/LibGfx)
    add_subdirectory(Libraries/LibHTTP)
    add_subdirectory(Libraries/LibImage)
    add_subdirectory(Libraries/LibIPC)
    add_subdirectory(Libraries/LibJSON)
    add_subdirectory(Libraries/LibLocale)
    add_subdirectory(Libraries/LibPackage)
    add_subdirectory(Libraries/LibServices)
    add_subdirectory(Libraries/LibSystem)
    add_subdirectory(Libraries/LibText)
    add_subdirectory(Libraries/LibTLS)
    add_subdirectory(Libraries/LibUnicode)
    add_subdirectory(Libraries/LibVT)
    add_subdirectory(Libraries/LibForeignFS)
    add_subdirectory(Libraries/LibVultureFS)
    add_subdirectory(Libraries/LibUI)
    add_subdirectory(Libraries/LibWindow)
    add_subdirectory(Applications)
    add_subdirectory(Libraries/LibDiskManagement)
    add_subdirectory(Services)
    add_subdirectory(Utilities)
    add_subdirectory(Tests/Network)
    add_subdirectory(Tests/Sound)
    add_subdirectory(Tests/UI)
    add_subdirectory(Tests/Userland)
    # The root filesystem skeleton: /etc and friends, and the macOS
    # hierarchy's directories even while nothing is in them (their modes
    # are in Meta/CMake/MakeBootArchive.cmake).
    install(DIRECTORY Base/ DESTINATION .)
    # The fonts for other languages and emoji, if they've been fetched (Meta/fetch-fonts.sh): they aren't in the repository.
    if(EXISTS ${CMAKE_BINARY_DIR}/../fonts)
        install(DIRECTORY ${CMAKE_BINARY_DIR}/../fonts/ DESTINATION System/Library/Fonts FILES_MATCHING PATTERN "*.ttf" PATTERN "*.ttc" PATTERN "*-VF.ttf" EXCLUDE)
        install(DIRECTORY ${CMAKE_BINARY_DIR}/../fonts/ DESTINATION usr/share/hunspell FILES_MATCHING PATTERN "en_US*")
    endif()
    # Which kind of machine this system is for: a disk may hold more than one.
    file(WRITE ${CMAKE_BINARY_DIR}/Architecture "${CMAKE_SYSTEM_PROCESSOR}\n")
    install(FILES ${CMAKE_BINARY_DIR}/Architecture DESTINATION System/Library/CoreServices)
    foreach(directory
            Applications Library Library/Preferences Library/LaunchDaemons Library/LaunchAgents
            System/Library/CoreServices System/Library/Frameworks System/Library/LaunchDaemons System/Library/LaunchAgents
            Users Users/Shared Volumes cores dev tmp
            usr/local usr/local/bin usr/share
            var var/root var/log var/run var/tmp var/db var/db/sudo)
        install(DIRECTORY DESTINATION ${directory})
    endforeach()
else()
    message(FATAL_ERROR "Unknown VULTURE_PART '${VULTURE_PART}'")
endif()
